\section{Introduction}

\begin{frame}{Research Question}

    \begin{block}{Question}
        Does intra-regional trade integration among ACP countries act as a
        \textbf{stumbling block} or \textbf{building block} for bilateral trade with the EU?
    \end{block}

    \bigskip

    \textbf{Why it matters}
    \begin{itemize}
        \item ACP countries pursue simultaneous goals: deepen regional integration \emph{and}
              maintain preferential EU access under EPAs
        \item These goals may conflict if regional integration diverts trade away from EU partners
        \item The post-Cotonou framework (2021) preserves the EPA architecture---design choices matter
    \end{itemize}

    \bigskip

    \textbf{Contribution}
    \begin{itemize}
        \item First empirical test of the stumbling-block hypothesis in the EU-ACP context
        \item Panel of 78 ACP countries $\times$ EU-27 $\times$ 1995--2021 with time-varying
              EPA treatment and REC membership
    \end{itemize}

\end{frame}

\begin{frame}{Preview of Main Findings}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Main result}
                IT share coefficient: \alert{$-0.564^{*}$} (s.e.\ $0.322$)\\[2pt]
                \textbf{Stumbling block confirmed}
            \end{block}
            \medskip
            \begin{block}{Magnitudes}
                At mean IT share (11.6\%): \alert{$-6.3\%$} bilateral trade\\
                At p90 IT share (25.5\%): \alert{$-13.4\%$} bilateral trade
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{block}{EPA moderation}
                EPA $\times$ IT share: \alert{$+1.048^{**}$}\\
                EPAs attenuate the stumbling block
            \end{block}
            \medskip
            \begin{block}{Direction}
                Driven by \alert{EU exports to ACP} ($-1.027^{***}$)\\
                ACP exports to EU: $-0.059$ (n.s.)\\[2pt]
                \small Consistent with import substitution
            \end{block}
        \end{column}
    \end{columns}

    \bigskip
    \small\textit{All seven RECs show negative IT share coefficients;
    five of six interpretable estimates are statistically significant.}

\end{frame}

\section{Institutional Background}

\begin{frame}{Institutional Background}

    \textbf{From Lom\'{e} to Cotonou to EPAs}
    \begin{itemize}
        \item \textbf{Lom\'{e} Convention} (1975--2000): non-reciprocal EU preferences for ACP exports
        \item \textbf{Cotonou Agreement} (2000): shift to reciprocal EPAs, WTO-compatible
        \item \textbf{EPAs}: free trade areas between the EU and each ACP regional group
    \end{itemize}

    \medskip

    \textbf{ACP Regional Groupings (7 RECs in sample)}

    \smallskip
    {\small
    \begin{tabular}{ll@{\qquad}ll}
        ECOWAS (16)    & West Africa      & EAC (5)       & East Africa \\
        SADC (8)       & Southern Africa  & COMESA (12)   & Eastern/Southern Africa \\
        CARIFORUM (15) & Caribbean        & Cent.\ Africa (7) & Central Africa \\
        PIF (15)       & Pacific          & & \\
    \end{tabular}}

    \medskip

    \textbf{EPA Implementation Timeline (provisional application dates)}
    \begin{itemize}
        \item 2008: CARIFORUM \quad 2009: PNG \quad 2012: ESA-4
        \item 2014: Cameroon, Fiji \quad 2016: SADC group \quad 2018: Mozambique \quad 2019: Comoros
        \item EAC: initialled but \textbf{never applied} within the 1995--2021 sample period
    \end{itemize}

\end{frame}

\section{Empirical Strategy}

\begin{frame}{Gravity Model Specification}

    \textbf{PPML Gravity Model} \citep{santos-silva2006}
    \[
        \underbrace{T_{ijt}}_{\text{EU-ACP trade}} = \exp\!\bigl(
            \beta_1 \underbrace{\text{ITshare}_{jt}}_{\text{key variable}} +
            \beta_2 \text{EPA}_{jt} +
            \beta_3 (\text{ITshare}_{jt} \times \text{EPA}_{jt}) +
            \gamma X_{ij} + \mu_{it} + \alpha_j + \theta_t
        \bigr) \cdot \varepsilon_{ijt}
    \]

    \medskip

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Fixed Effects}
            \begin{itemize}
                \item $\mu_{it}$: Exporter $\times$ Year FE\\
                      \small absorbs all EU time-varying factors
                \item $\alpha_j$: ACP country FE\\
                      \small absorbs time-invariant ACP characteristics
                \item $\theta_t$: Year FE\\
                      \small absorbs common time shocks
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Why not Importer $\times$ Year FE?}
            \begin{itemize}
                \item ITshare and EPA vary only at the ACP country $\times$ year level
                \item Importer $\times$ Year FEs would fully \emph{absorb} both variables
                \item Solution: ACP country FE $+$ Year FE preserves identification while
                      controlling for common time shocks
            \end{itemize}
        \end{column}
    \end{columns}

    \medskip
    \textbf{Identification:} within-ACP-country variation in IT share over time,
    across 27 EU partner-year observations per country-year.

\end{frame}

    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \textbf{Key finding:} stumbling block is entirely an \alert{EU-to-ACP} phenomenon
            \begin{itemize}
                \item ACP exports to EU: unaffected ($-0.059$, n.s.)
                \item EU exports to ACP: strongly negative ($-1.027^{***}$)
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \textbf{Interpretation:} consistent with \textbf{import substitution}
            \begin{itemize}
                \item Rising intra-bloc integration displaces EU goods in ACP domestic markets
                \item ACP primary-commodity exports to the EU face little intra-bloc competition
            \end{itemize}
        \end{column}
    \end{columns}

\section{Conclusion}

\begin{frame}{Conclusion}

    \begin{block}{Main finding}
        Strong, consistent \textbf{stumbling-block} effect of intra-ACP regional integration
        on EU-ACP bilateral trade---robust across specifications, RECs, and subsamples.
    \end{block}

    \medskip

    \textbf{What drives it?}
    \begin{itemize}
        \item Effect is entirely in EU exports to ACP markets ($-1.027^{***}$), not
              ACP exports to the EU ($-0.059$, n.s.)
        \item Consistent with import substitution: rising intra-bloc integration displaces
              EU goods in ACP domestic markets
        \item EPAs moderate but do not eliminate the effect at observed integration levels
    \end{itemize}

    \medskip

    \textbf{Policy implications}
    \begin{itemize}
        \item EPA conditionality design should account for the trade-diverting phase of integration
        \item EPAs and regional integration are most synergistic when advancing \emph{simultaneously}
        \item At SADC mean IT share (27\%): estimated $-14.2\%$ bilateral trade with the
              EU---a significant development consequence for smaller SADC members
    \end{itemize}

\end{frame}
